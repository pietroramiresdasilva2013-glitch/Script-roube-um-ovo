local function SetStatus(text, color)
    Status.Text = "Status: " .. text
    Status.TextColor3 = color or Color3.fromRGB(160, 160, 160)
end

local function GetRarityScore(name)
    name = name:lower()
    if name:find("divine") then return 10
    elseif name:find("secret") then return 9
    elseif name:find("myth") then return 8
    elseif name:find("legend") then return 7
    elseif name:find("epic") then return 6
    elseif name:find("rare") then return 5
    elseif name:find("uncommon") then return 4
    else return 3
    end
end

local function FindBestEgg()
    local bestEgg = nil
    local bestScore = 0

    for _, obj in pairs(workspace:GetDescendants()) do
        if obj:IsA("Model") or obj:IsA("BasePart") then
            local name = obj.Name:lower()
            if name:find("egg") or name:find("ovo") then
                local prompt = obj:FindFirstChildWhichIsA("ProximityPrompt", true)
                if prompt and prompt.Enabled then
                    local score = GetRarityScore(obj.Name)
                    if score > bestScore then
                        bestScore = score
                        bestEgg = obj
                    end
                end
            end
        end
    end
    return bestEgg
end

local function Teleport(cf)
    if HumanoidRootPart and cf then
        HumanoidRootPart.CFrame = cf + Vector3.new(0, 3, 0)
    end
end

-- Loop do Auto Farm
spawn(function()
    while true do
        if Config.AutoFarm and Config.HomeCFrame then
            local egg = FindBestEgg()
            if egg then
                SetStatus("Indo no ovo mais raro...", Color3.fromRGB(0, 220, 255))
                Teleport(egg:GetPivot())
                task.wait(0.45)

                local prompt = egg:FindFirstChildWhichIsA("ProximityPrompt", true)
                if prompt then
                    fireproximityprompt(prompt)
                    SetStatus("Ovo pego! Voltando...", Color3.fromRGB(0, 255, 120))
                    task.wait(0.35)
                    Teleport(Config.HomeCFrame)
                    task.wait(1.2)
                end
            else
                SetStatus("Nenhum ovo encontrado...", Color3.fromRGB(255, 180, 50))
                task.wait(1.5)
            end
        end
        task.wait(0.6)
    end
end)

-- Arrastar
local dragging, dragStart, startPos
Header.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = Main.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then dragging = false end
        end)
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = input.Position - dragStart
        Main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)
